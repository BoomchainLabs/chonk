# token route
cat > app/api/token/route.ts <<'EOL'
import { NextResponse } from "next/server";
import axios from "axios";
const MINT = process.env.NEXT_PUBLIC_TOKEN_MINT;
export async function GET() {
  try { const res = await axios.get(`https://api.dexscreener.com/latest/dex/tokens/${MINT}`); return NextResponse.json(res.data); }
  catch { return NextResponse.json({error:"failed to fetch token info"}); }
}
EOL
# holders route
cat > app/api/holders/route.ts <<'EOL'
import { NextResponse } from "next/server";
import { Connection, PublicKey } from "@solana/web3.js";
export async function GET() {
  const connection = new Connection(process.env.NEXT_PUBLIC_RPC_URL!);
  const mint = new PublicKey(process.env.NEXT_PUBLIC_TOKEN_MINT);
  const largest = await connection.getTokenLargestAccounts(mint);
  return NextResponse.json(largest.value);
}
EOL
# transfers route
cat > app/api/transfers/route.ts <<'EOL'
import { NextResponse } from "next/server";
import { Connection, PublicKey } from "@solana/web3.js";
export async function GET() {
  const connection = new Connection(process.env.NEXT_PUBLIC_RPC_URL!);
  const mint = new PublicKey(process.env.NEXT_PUBLIC_TOKEN_MINT);
  const sigs = await connection.getSignaturesForAddress(mint,{limit:20});
  return NextResponse.json(sigs);
}
EOL
